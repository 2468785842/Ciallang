cmake_minimum_required(VERSION 3.25)
project(CiallangEmbeddingSmoke LANGUAGES CXX)

# The library must provide its own C++20 requirement without changing this default.
set(CMAKE_CXX_STANDARD 17)
set(ENABLE_TESTS ON)
set(BUILD_EXECUTABLE ON)
set(ENABLE_FUZZ ON)
set(CMAKE_DISABLE_FIND_PACKAGE_argparse TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_Catch2 TRUE)
get_filename_component(CIAL_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_subdirectory("${CIAL_SOURCE_DIR}" ciallang)
foreach(forbidden_target cial cial_bc CiallangTests fuzz_lexer fuzz_parser)
    if(TARGET ${forbidden_target})
        message(FATAL_ERROR "Embedding unexpectedly created ${forbidden_target}")
    endif()
endforeach()
if(NOT CMAKE_CXX_STANDARD EQUAL 17)
    message(FATAL_ERROR "Ciallang changed the parent C++ standard")
endif()
add_executable(embedding_smoke main.cpp)
target_link_libraries(embedding_smoke PRIVATE Ciallang::Ciallang)
enable_testing()
add_test(NAME embedding_smoke COMMAND embedding_smoke)
