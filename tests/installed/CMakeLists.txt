cmake_minimum_required(VERSION 3.25)
project(CiallangInstalledHost LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_DISABLE_FIND_PACKAGE_argparse TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_Catch2 TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_Python3 TRUE)
find_package(Ciallang CONFIG REQUIRED)
add_executable(host main.cpp)
target_link_libraries(host PRIVATE Ciallang::Ciallang)
if(MSVC)
    target_compile_options(host PRIVATE /utf-8)
endif()
enable_testing()
add_test(NAME installed_host COMMAND host)
